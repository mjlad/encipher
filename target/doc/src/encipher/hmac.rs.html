8/// Combines key and start_from into a 16-byte secret.
9/// Concatenation is used instead of addition to avoid collisions
10/// (different inputs producing the same sum).
11/// to_be_bytes ensures consistent byte order across all platforms (Big Endian).
12fn build_secret(key: u64, start_from: usize) -> [u8; 16] {
13    let mut secret = [0u8; 16];
14    secret[..8].copy_from_slice(&key.to_be_bytes());
15    secret[8..].copy_from_slice(&(start_from as u64).to_be_bytes());
16    secret
17}

20/// Produces an HMAC-SHA256 signature for the given ciphertext.
21pub fn sign(cipher_text: &str, start_from: usize, key: u64) -> String {
22    let mut mac = HmacSha256::new_from_slice(
23        &build_secret(key, start_from))
24        .expect("HMAC init failed");
25    mac.update(cipher_text.as_bytes());
26    URL_SAFE_NO_PAD.encode(mac.finalize().into_bytes())
27}

30/// Verifies the signature using constant-time comparison to prevent timing attacks.
31pub fn verify(cipher_text: &str, start_from: usize, key: u64, signature: &str) -> bool {
32    let mut mac = HmacSha256::new_from_slice(
33        &build_secret(key, start_from))
34        .expect("HMAC init failed");
35    mac.update(cipher_text.as_bytes());
36
37    let Ok(sig_bytes) = URL_SAFE_NO_PAD.decode(signature) else {
38        return false;
39    };
40
41    mac.verify_slice(&sig_bytes).is_ok()
42}
