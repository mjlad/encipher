7/// A substitution cipher lookup table for a single encryption layer.
8///
9/// Contains two precomputed tables:
10/// - `enc`: maps plaintext bytes to ciphertext bytes.
11/// - `dec`: maps ciphertext bytes back to plaintext bytes.
12#[derive(Debug)]
13pub struct CipherList {
14    pub enc: [u8; 256],
15    pub dec: [u8; 256],
16}
17
18impl CipherList {
19    /// Builds an encrypt/decrypt lookup table using a seeded RNG and a step offset.
20    ///
21    /// `enc` and `dec` are precomputed substitution tables to guarantee
22    /// O(1) encryption and decryption per byte at runtime.
23    pub fn new(rng: &mut ChaCha20Rng, step: u8) -> Self {
24        let mut forward: [u8; 256] = core::array::from_fn(|i| i as u8);
25        forward.shuffle(rng);
26
27        let step = step as usize;
28        let mut reverse = [0u8; 256];
29
30        for (i, &v) in forward.iter().enumerate() {
31            reverse[v as usize] = i as u8;
32        }
33
34        let enc = core::array::from_fn(|i| {
35            forward[(reverse[i] as usize + step) % 256]
36        });
37
38        let dec = core::array::from_fn(|i| {
39            forward[(256 + reverse[i] as usize - step) % 256]
40        });
41
42        
43        #[cfg(debug_assertions)]
44        for byte in 0u8..=255 {
45            debug_assert_eq!(
46                dec[enc[byte as usize] as usize],
47                byte,
48                "List error when byte={}", byte
49            );
50        }
51
52        CipherList { enc, dec }
53    }
54}

58/// Generates 100 substitution lists from a fixed key.
59/// Count is fixed in the type signature for performance
60/// and to avoid unnecessary heap allocation.
61pub fn generate_lists(key: u64, step: u8) -> Box<[CipherList; 100]> {
62    let mut rng = ChaCha20Rng::seed_from_u64(key);
63    let lists: Vec<CipherList> = (0..100).map(|_| CipherList::new(&mut rng, step)).collect();
64    lists.try_into().unwrap()
65}
